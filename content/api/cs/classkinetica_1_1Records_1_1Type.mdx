<h2 className="groupheader">Detailed Description</h2>
<div className="textblock">
<p>Immutable collection of metadata about a <a className="el" href="/content/api/cs/classkinetica_1_1Kinetica" title="API to talk to Kinetica Database.">Kinetica</a> type. </p>
<p>Equivalent to Rust's <a className="el" href="/content/api/cs/classkinetica_1_1Records_1_1Type" title="Immutable collection of metadata about a Kinetica type.">Type</a> struct.</p>
<p>This class contains all the information needed to create and encode records of this type, including schema information, column metadata, and key indices.</p>
<div className="fragment">
<div className="line"><span className="comment">// From existing table</span></div>
<div className="line">var recordType = await <a className="code hl_function" href="#afbcc78dd6ffff5854955b8a6682fa113">Type</a>.FromTableAsync(<a className="code hl_namespace" href="/content/api/cs/namespacekinetica">kinetica</a>, <span className="stringliteral">&quot;my\_table&quot;</span>);</div>
<div className="line"> </div>
<div className="line"><span className="comment">// Or create programmatically</span></div>
<div className="line">var recordType = <a className="code hl_function" href="#afbcc78dd6ffff5854955b8a6682fa113">Type</a>.Builder(<span className="stringliteral">&quot;my\_record&quot;</span>)</div>
<div className="line"> .AddIntColumn(<span className="stringliteral">&quot;id&quot;</span>).PrimaryKey().ShardKey()</div>
<div className="line"> .AddStringColumn(<span className="stringliteral">&quot;name&quot;</span>)</div>
<div className="line"> .AddDoubleColumn(<span className="stringliteral">&quot;score&quot;</span>).Nullable()</div>
<div className="line"> .Build();</div>
<div className="line"> </div>
