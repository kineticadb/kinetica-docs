<div className="docblock">
<p>Builder for creating Kinetica connection pools with sensible defaults.</p>
<h2 id="example">Example</h2>
<div>

```rust
use gpudb_api::sqlx_support::KineticaPoolOptions;
use std::time::Duration;

let pool = KineticaPoolOptions::new()
    .host("localhost")
    .port(5432)
    .username("admin")
    .password("password")
    .database("postgres")
    .max_connections(10)
    .min_connections(2)
    .acquire_timeout(Duration::from_secs(30))
    .connect()
    .await?;
```
